{{- /* Rende un'immagine da assets/ in WebP. Contesto: dict "percorso" "img/x.jpg" "alt" "..." "larghezza" 800 Con "ritaglio" (es. "640x400") ritaglia a misura fissa: card tutte uguali. Con "sizes" si dichiara la larghezza resa: genera varianti più piccole (srcset) così i telefoni non scaricano pixel inutili. Con "prioritaria" true l'immagine è sopra la piega (LCP): niente lazy loading né skeleton, e fetchpriority alta. */ -}} {{- $ris := resources.Get .percorso -}} {{- if $ris -}} {{- $q := .qualita | default 80 -}} {{- $img := "" -}} {{- $varianti := slice -}} {{- if .ritaglio -}} {{- /* "ancora" sposta il punto di ritaglio (es. "Top" per non tagliare le teste) */ -}} {{- $img = $ris.Fill (printf "%s %s webp q%d" .ritaglio (.ancora | default "Smart") $q) -}} {{- $dim := split .ritaglio "x" -}} {{- $varianti = $varianti | append ($ris.Fill (printf "%dx%d %s webp q%d" (div (int (index $dim 0)) 2) (div (int (index $dim 1)) 2) (.ancora | default "Smart") $q)) -}} {{- else -}} {{- $larg := .larghezza | default 800 -}} {{- if gt $larg $ris.Width }}{{ $larg = $ris.Width }}{{ end -}} {{- $img = $ris.Resize (printf "%dx webp q%d" $larg $q) -}} {{- if gt $img.Width 900 -}} {{- $varianti = $varianti | append ($ris.Resize (printf "%dx webp q%d" (int (mul $img.Width 0.66)) $q)) -}} {{- end -}} {{- if gt $img.Width 500 -}} {{- $varianti = $varianti | append ($ris.Resize (printf "%dx webp q%d" (div $img.Width 2) $q)) -}} {{- end -}} {{- end -}} {{- $srcset := slice -}} {{- range $varianti }}{{ $srcset = $srcset | append (printf "%s %dw" .RelPermalink .Width) }}{{ end -}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink $img.Width) -}} {{- end -}}
